# Main change detection SQL

INSERT INTO customer_changes
SELECT
    CustomerId,

    firstNameEncrypted,
    firstNameHashed,
    CASE
        WHEN firstNameHashed IS DISTINCT FROM prevFirstNameHash
        THEN TRUE
        ELSE CAST(NULL AS BOOLEAN)
    END AS firstNameChanged,

    lastNameEncrypted,
    lastNameHashed,
    CASE
        WHEN lastNameHashed IS DISTINCT FROM prevLastNameHash
        THEN TRUE
        ELSE CAST(NULL AS BOOLEAN)
    END AS lastNameChanged,

    DOBEncrypted,
    DOBHashed,
    CASE
        WHEN DOBHashed IS DISTINCT FROM prevDOBHash
        THEN TRUE
        ELSE CAST(NULL AS BOOLEAN)
    END AS DOBChanged

FROM (
    SELECT
        *,

        LAG(CustomerId, 1) OVER (
            PARTITION BY CustomerId
            ORDER BY proc_time
        ) AS prevCustomerId,

        LAG(firstNameHashed, 1) OVER (
            PARTITION BY CustomerId
            ORDER BY proc_time
        ) AS prevFirstNameHash,

        LAG(lastNameHashed, 1) OVER (
            PARTITION BY CustomerId
            ORDER BY proc_time
        ) AS prevLastNameHash,

        LAG(DOBHashed, 1) OVER (
            PARTITION BY CustomerId
            ORDER BY proc_time
        ) AS prevDOBHash

    FROM customer_source
) AS history

WHERE prevCustomerId IS NOT NULL
  AND (
       firstNameHashed IS DISTINCT FROM prevFirstNameHash
    OR lastNameHashed IS DISTINCT FROM prevLastNameHash
    OR DOBHashed IS DISTINCT FROM prevDOBHash
  );
